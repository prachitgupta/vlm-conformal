    pid=$(echo $pid | tr -d ',')
    mem=$(echo $mem | tr -d ' MiB,')
    user=$(ps -p $pid -o user= 2>/dev/null || echo "unknown")
    cmd=$(ps -p $pid -o comm= 2>/dev/null || echo "unknown")
    echo "$user | PID: $pid | Memory: $mem MiB | Process: $cmd"
done
